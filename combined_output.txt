"""
Celery application configuration.

CRITICAL: This module creates the single shared Celery instance.
It must be imported before any tasks (manual or Dash callbacks) are defined.
"""

import os
from celery import Celery

# Redis URL from environment or default for local development
REDIS_URL = os.environ.get("REDIS_URL", "redis://localhost:6379/0")

# Create the single Celery app instance
# This MUST be created before Dash app or any tasks are imported
celery_app = Celery(
    "app",
    broker=REDIS_URL,
    backend=REDIS_URL,
)

# Celery configuration
celery_app.conf.update(
    task_serializer="json",
    accept_content=["json"],
    result_serializer="json",
    timezone="UTC",
    enable_utc=True,
    # Task discovery - we'll also manually ensure dash callbacks are registered
    # by importing them in the worker entry point
    task_track_started=True,
    result_extended=True,
)

the background callbacks as Celery tasks. The callbacks are registered
when this module is imported and the @callback decorators execute.

"""
Dash callbacks - both regular and background.

CRITICAL FOR BACKGROUND CALLBACKS:
When using background=True, Dash registers the callback function as a Celery task.
This happens at import time when the @callback decorator executes.

For Celery workers to know about these tasks, they MUST import this module
(which happens when app.dash_app is imported).
"""

import time
import plotly.graph_objects as go
from dash import callback, Input, Output, State

# Import the manager for background callbacks
from app.dash_app.app import background_callback_manager


def make_progress_figure(current: int, total: int) -> go.Figure:
    """Create a progress bar figure."""
    fig = go.Figure(
        data=[go.Bar(x=[current], orientation="h", marker_color="green")]
    )
    fig.update_xaxes(range=[0, total], title="Progress")
    fig.update_yaxes(showticklabels=False)
    fig.update_layout(height=100, margin=dict(t=20, b=40, l=40, r=40))
    return fig


# =============================================================================
# BACKGROUND CALLBACK - This is the key part
# =============================================================================
@callback(
    output=Output("task-status", "children"),
    inputs=Input("run-task-btn", "n_clicks"),
    background=True,  # This makes it a Celery task
    manager=background_callback_manager,
    running=[
        # While running: disable run button, enable cancel button
        (Output("run-task-btn", "disabled"), True, False),
        (Output("cancel-btn", "disabled"), False, True),
        # Show progress graph while running
        (Output("progress-graph", "style"), {"display": "block"}, {"display": "none"}),
    ],
    cancel=[Input("cancel-btn", "n_clicks")],
    progress=Output("progress-graph", "figure"),
    progress_default=make_progress_figure(0, 10),
    prevent_initial_call=True,
)
def run_background_task(set_progress, n_clicks):
    """
    Background callback that runs as a Celery task.

    The set_progress function allows updating the UI while the task runs.
    The 'running' parameter controls component states during execution.
    """
    total_steps = 10

    for i in range(total_steps):
        # Simulate work
        time.sleep(1)
        # Update progress
        set_progress(make_progress_figure(i + 1, total_steps))

    return f"Task completed! (clicked {n_clicks} times)"


# =============================================================================
# REGULAR CALLBACK - For comparison, this runs synchronously
# =============================================================================
@callback(
    Output("output-text", "children"),
    Input("input-text", "value"),
    prevent_initial_call=True,
)
def update_output(value):
    """Regular synchronous callback."""
    if value:
        return f"You typed: {value}"
    return ""

"""
FastAPI application that mounts the Dash app and exposes API endpoints.

This is the main entry point for the web application.
"""

from fastapi import FastAPI, HTTPException
from fastapi.middleware.wsgi import WSGIMiddleware
from pydantic import BaseModel
from celery.result import AsyncResult

# Import the Dash app - this also triggers callback registration
from app.dash_app import dash_app

# Import Celery app and tasks for the API endpoints
from app.celery_config import celery_app
from app import tasks


# Create FastAPI app
app = FastAPI(
    title="FastAPI + Dash + Celery Demo",
    description="Demonstrates shared Celery between FastAPI endpoints and Dash background callbacks",
)


# =============================================================================
# API Models
# =============================================================================
class TaskRequest(BaseModel):
    duration: int = 5


class DataRequest(BaseModel):
    data: dict


class TaskResponse(BaseModel):
    task_id: str
    status: str


class TaskStatusResponse(BaseModel):
    task_id: str
    status: str
    result: dict | None = None
    progress: dict | None = None


# =============================================================================
# FastAPI Endpoints that use Celery tasks
# =============================================================================
@app.get("/")
async def root():
    """Root endpoint with links to other parts of the app."""
    return {
        "message": "FastAPI + Dash + Celery Demo",
        "endpoints": {
            "dash_app": "/dash/",
            "api_docs": "/docs",
            "start_task": "/api/tasks/start",
            "task_status": "/api/tasks/{task_id}",
        }
    }


@app.post("/api/tasks/start", response_model=TaskResponse)
async def start_task(request: TaskRequest):
    """
    Start a long-running Celery task.

    This demonstrates calling a manually-defined Celery task from FastAPI.
    """
    result = tasks.long_running_task.delay(duration=request.duration)
    return TaskResponse(task_id=result.id, status="PENDING")


@app.post("/api/tasks/process", response_model=TaskResponse)
async def process_data(request: DataRequest):
    """Start a data processing task."""
    result = tasks.process_data.delay(data=request.data)
    return TaskResponse(task_id=result.id, status="PENDING")


@app.get("/api/tasks/{task_id}", response_model=TaskStatusResponse)
async def get_task_status(task_id: str):
    """
    Get the status of a Celery task.

    Works for both manually-defined tasks and Dash background callbacks.
    """
    result = AsyncResult(task_id, app=celery_app)

    response = TaskStatusResponse(
        task_id=task_id,
        status=result.status,
    )

    if result.ready():
        response.result = result.result
    elif result.status == "PROGRESS":
        response.progress = result.info

    return response


@app.delete("/api/tasks/{task_id}")
async def cancel_task(task_id: str):
    """Cancel a running Celery task."""
    celery_app.control.revoke(task_id, terminate=True)
    return {"task_id": task_id, "status": "REVOKED"}


# =============================================================================
# Mount Dash app
# =============================================================================
# Dash is a Flask (WSGI) app, so we use WSGIMiddleware
app.mount("/dash", WSGIMiddleware(dash_app.server))

"""
Manual Celery tasks for FastAPI endpoints.

These are the traditional @celery_app.task decorated functions
that FastAPI endpoints will call directly.
"""

import time
from app.celery_config import celery_app


@celery_app.task(bind=True)
def long_running_task(self, duration: int = 5) -> dict:
    """
    Example long-running task that can be called from FastAPI endpoints.

    Args:
        duration: How long to simulate work (seconds)

    Returns:
        dict with task result
    """
    for i in range(duration):
        # Update task state for progress tracking
        self.update_state(
            state="PROGRESS",
            meta={"current": i + 1, "total": duration}
        )
        time.sleep(1)

    return {
        "status": "completed",
        "duration": duration,
        "message": f"Task completed after {duration} seconds"
    }


@celery_app.task
def process_data(data: dict) -> dict:
    """
    Example data processing task.
    """
    # Simulate processing
    time.sleep(2)
    return {
        "processed": True,
        "input_keys": list(data.keys()),
        "result": "Data processed successfully"
    }
===== FILE: celery_worker.py =====
"""
Celery worker entry point.

THIS IS THE KEY FILE THAT SOLVES THE TASK REGISTRATION PROBLEM.

When starting the Celery worker, point it at this module:
    celery -A celery_worker.celery_app worker --loglevel=info

This module imports:
1. The shared Celery app instance
2. The manual tasks module (registers @celery_app.task decorated functions)
3. The Dash app module (registers background callbacks as Celery tasks)

The order matters:
1. celery_app must exist first
2. Then tasks that depend on it can be imported
3. Dash app import triggers callback decoration, which registers them as tasks

Without importing the Dash app here, the worker would never know about
the background callbacks because they're registered dynamically when
the @callback decorators execute.
"""

# Step 1: Import the Celery app instance
from app.celery_config import celery_app

# Step 2: Import manual tasks - these use @celery_app.task decorator
from app import tasks  # noqa: F401

# Step 3: CRITICAL - Import the Dash app to register background callbacks
# When app.dash_app is imported, it imports callbacks.py, which executes
# the @callback decorators with background=True, registering them as Celery tasks
from app.dash_app import dash_app  # noqa: F401

# The celery_app is now fully configured with all tasks registered
# Export it so the celery command can find it
__all__ = ["celery_app"]
